How to use Most Common Events:
--------------------------------

 1. click — Sabse common(simple click per active)

const btn = document.querySelector("#btn");
btn.addEventListener("click", () => {
  alert("Button click hua! 🔥");
});
_____________________________________________________________________________________

 2. input — Live typing pakdo...

const input = document.querySelector("#search");
input.addEventListener("input", (e) => {
  console.log(e.target.value); // jo type ho raha hai live
});
// e.target.value = input mein jo likha hai wo milta hai adject words...
_____________________________________________________________________________________

 3. submit — Form submit rokna...
const form = document.querySelector("#myForm");

form.addEventListener("submit", (e) => {
  e.preventDefault(); // ← Page reload rokta hai! Bahut important!
  console.log("Form submit hua!");
});
//e.preventDefault() — ye likhna mat bhoolna submit pe ⚠️ kuki ye reload ni hone denga form jab tk bhar na jaye ..
_____________________________________________________________________________________

 4. keydown — Keyboard key pakdo konsi key press ki bo
document.addEventListener("keydown", (e) => {
  console.log(e.key);     // "Enter", "a", "Escape" etc.
  console.log(e.keyCode); // 13, 65, 27 etc.
  if (e.key === "Enter") {
    console.log("Enter dabaya! ✅");
  }
  if (e.key === "Escape") {
    console.log("Modal band karo!");
  }
});
_____________________________________________________________________________________

5. mouseover & mouseout — Hover effect ke liye 
const box = document.querySelector(".box");
box.addEventListener("mouseover", () => {
  box.style.backgroundColor = "yellow"; // mouse aaya
});
box.addEventListener("mouseout", () => {
  box.style.backgroundColor = "white";  // mouse gaya
});
_____________________________________________________________________________________

 6. scroll — Page scroll detect karo ki scroll hua hai 
window.addEventListener("scroll", () => {
  const scrollY = window.scrollY; // kitna scroll hua
  console.log(scrollY);

  if (scrollY > 200) {
    document.querySelector(".navbar").classList.add("sticky");
  }
});
_____________________________________________________________________________________

7. focus & blur — Input active/inactive
const input = document.querySelector("#name");
input.addEventListener("focus", () => {
  input.style.border = "2px solid blue"; // active hua
});
input.addEventListener("blur", () => {
  input.style.border = "2px solid gray"; // inactive hua
});
_____________________________________________________________________________________

 8. load — Page load hone ke baad
window.addEventListener("load", () => {
  console.log("Page poora load ho gaya! ✅");
  // Yahan se safe hai DOM use karna
});
_____________________________________________________________________________________

~~~ Event Object (e) — Bahut kaam aata hai isko yd kr lo*****
btn.addEventListener("click", (e) => {
  console.log(e.target);      // Kis element pe click hua
  console.log(e.type);        // "click"
  console.log(e.clientX);     // Mouse X position
  console.log(e.clientY);     // Mouse Y position
  
  e.preventDefault();         // Default action rokta hai
  e.stopPropagation();        // Event upar jaane se rokta hai
});

  <li>Item 1</li>

                // ❌ Galat tarika — har li pe alag listener

    console.log(e.target.textContent + " click hua!");
  }
});
